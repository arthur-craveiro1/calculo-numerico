ESTRUTURAS TIPICAS EM METODOS NUMERICOS

def bissecao(f, a, b, tol, max_iter):
    # Metodo da bissecao: reduz o intervalo pela metade a cada iteracao
    for i in range(max_iter):
        m = (a + b) / 2
        if abs(f(m)) < tol:
            return m
        elif f(a) * f(m) < 0:
            b = m
        else:
            a = m
    return m

def newton_raphson(f, df, x0, tol, max_iter):
    # Usa a derivada de f para convergir mais rapido ate a raiz
    x = x0
    for i in range(max_iter):
        x_novo = x - f(x) / df(x)
        if abs(x_novo - x) < tol:
            return x_novo
        x = x_novo
    return x

def trapezio(f, a, b, n):
    # Integracao numerica pela regra do trapezio
    h = (b - a) / n
    soma = (f(a) + f(b)) / 2
    for i in range(1, n):
        soma += f(a + i * h)
    return soma * h
